# SpeechLLm — runtime dependencies for the VieNeu-TTS service.
#
# This file used to carry ~160 commented-out lines plus the entire Coqui stack
# (TTS==0.22.0, torch, torchaudio, trainer, coqpit, gruut*, elevenlabs,
# faster-whisper, ollama). None of it was reachable: the live import chain is
# exactly two modules —
#
#     api_server.py -> src/services/vieneu_client.py
#
# (src/services/tts_router.py existed here too until 11-09-2026, as a
# single-provider passthrough used only by the now-deleted POST /synthesize.)
#
# and at the same time the file was MISSING fastapi and uvicorn, which
# api_server.py imports on line 7-11. So it installed two gigabytes of dead
# weight and still could not start the server.
#
# Verified: these versions are the ones the service was measured and accepted
# under on 10-09-2026 (Python 3.12.14, CPU, no GPU).

# ── Engine ───────────────────────────────────────────────────────────────────
# PINNED, and the pin is load-bearing rather than hygiene. The calling
# convention changed between generations: `mode: turbo` (v2) needs the [gpu]
# extra, and `voice=` accepts a different shape than it does in v3turbo. An
# unpinned upgrade does not fail loudly — it degrades to the preset voice with
# a log line still reading "cloned". See src/services/vieneu_client.py.
#
# Pulls in onnxruntime, numpy, soundfile, librosa and gradio transitively.
# Deliberately NOT pinned here: letting vieneu resolve its own tree avoids the
# conflicts that over-pinning creates.
vieneu==3.6.4

# soundfile and numpy arrive transitively through vieneu (see above), but
# vieneu_client.py now imports both directly (encoding each stream chunk to
# Opus/WAV, building the float32 buffers) — the file's own rule is that a
# direct import gets declared even when a transitive install would satisfy
# it today. Pinned to whatever vieneu==3.6.4 already resolved them to in the
# tts env, not chosen independently, so there is nothing here for pip to
# conflict over.
soundfile==0.14.0
numpy==2.5.3

# ── Web service ──────────────────────────────────────────────────────────────
# Declared explicitly even though vieneu happens to pull all four in today.
# api_server.py imports them directly, and a dependency that arrives by accident
# is one `vieneu` release away from disappearing.
fastapi==0.141.1
uvicorn==0.52.4
pydantic==2.13.5
PyYAML==6.0.3

# ── S3 voice bucket (D5d) ───────────────────────────────────────────────────
# SpeechLLm reads reference voices from the private voice bucket (no CloudFront)
# via IAM read-only on prefix. Local dev keeps plain files under voices/*.wav
# when VOICE_BUCKET is unset. boto3 is the only S3 client here — the image is
# python:3.12-slim, not Lambda base, so nothing is preinstalled.
boto3>=1.34.0
